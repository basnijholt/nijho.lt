{{- /* Returns the page title. A taxonomy page without a content file is titled after its plural ("Publication types"), and a publication type after its display name. Hugo titles a term page without a content file after whichever spelling of the term it reads first, which changes between builds (AI or ai); such a page gets the first spelling with capitals, else the first spelling, in sorted order. */ -}} {{- if .File }} {{- return .Title }} {{- end }} {{- if eq .Kind "taxonomy" }} {{- return (humanize .Data.Plural) }} {{- end }} {{- if ne .Kind "term" }} {{- return .Title }} {{- end }} {{- if eq .Data.Plural "publication_types" }} {{- return (partial "functions/publication_type.html" .Data.Term) }} {{- end }} {{- $term := urlize .Data.Term }} {{- $spellings := slice }} {{- range .Pages }} {{- range index .Params $.Data.Plural }} {{- if eq (urlize .) $term }} {{- $spellings = $spellings | append . }} {{- end }} {{- end }} {{- end }} {{- $spellings = $spellings | uniq | sort }} {{- $capitalized := slice }} {{- range $spellings }} {{- if ne . (lower .) }} {{- $capitalized = $capitalized | append . }} {{- end }} {{- end }} {{- with $capitalized | default $spellings }} {{- return (index . 0 | title) }} {{- end }} {{- return .Title -}}
